-- TAB 1: Combat & Aimbot
local CombatTab = Window:Tab({ Title = "Combat / Aimbot", Icon = "swords" })
CombatTab:Section({ Title = "Aimbot & Targeting" })

CombatTab:Toggle({
    Title = "Camera Lock Aimbot (Lock onto Nearest Enemy)",
    Desc = "Automatically directs your camera to the closest enemy's upper torso",
    Value = false,
    Callback = function(state)
        _G.CameraAimbot = state
        RunService.RenderStepped:Connect(function()
            if not _G.CameraAimbot then return end
            local closestPlayer = nil
            local shortestDistance = math.huge
            
            for _, p in pairs(Players:GetPlayers()) do
                if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") and p.Character:FindFirstChild("Humanoid") and p.Character.Humanoid.Health > 0 then
                    local hrp = p.Character.HumanoidRootPart
                    local dist = (hrp.Position - LocalPlayer.Character.HumanoidRootPart.Position).Magnitude
                    if dist < shortestDistance then
                        shortestDistance = dist
                        closestPlayer = p
                    end
                end
            end
            
            if closestPlayer and closestPlayer.Character and closestPlayer.Character:FindFirstChild("HumanoidRootPart") then
                local targetPos = closestPlayer.Character.HumanoidRootPart.Position
                Camera.CFrame = CFrame.new(Camera.CFrame.Position, targetPos)
            end
        end)
    end
})

CombatTab:Toggle({
    Title = "ESP Enemy Highlights",
    Desc = "Puts glowing red outlines on enemies",
    Value = false,
    Callback = function(state)
        _G.KD_ESP = state
        RunService.RenderStepped:Connect(function()
            if not _G.KD_ESP then return end
            for _, p in pairs(Players:GetPlayers()) do
                if p ~= LocalPlayer and p.Character then
                    local hl = p.Character:FindFirstChild("CrownKDHighlight") or Instance.new("Highlight")
                    hl.Name = "CrownKDHighlight"
                    hl.FillColor = Color3.fromRGB(255, 0, 0)
                    hl.OutlineColor = Color3.fromRGB(255, 255, 255)
                    hl.Parent = p.Character
                end
            end
        end)
    end
})

CombatTab:Slider({
    Title = "Knife Hitbox Expander",
    Min = 2,
    Max = 20,
    Default = 2,
    Callback = function(size)
        _G.HitboxSize = size
        task.spawn(function()

                    for _, p in pairs(Players:GetPlayers()) do
                        if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then

                            p.Character.HumanoidRootPart.Transparency = 0.7

            end
        end)
    end
})
